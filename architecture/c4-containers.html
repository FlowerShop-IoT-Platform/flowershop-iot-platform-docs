C4 Level 2 — Containers

How the platform’s runtime containers communicate.

C4Container
  title FlowerShop IoT Platform — Containers

  Person(customer, "Customer")
  Person(vendor, "Vendor")
  Person(admin, "Platform Admin")

  Container_Boundary(platform, "FlowerShop IoT Platform") {
    Container(api, "REST API", ".NET 8 / ASP.NET Core", "CQRS, multi-tenant, 23 controllers, MediatR pipeline")
    Container(adminPortal, "Admin Portal", "ASP.NET Razor Pages :5001", "Vendor management, analytics, vase registration")
    Container(vendorPortal, "Vendor Portal", "ASP.NET Razor Pages :5002", "Bouquet catalog, order management, analytics dashboard")
    Container(customerApp, "Customer Web App", "ASP.NET MVC :5003", "Browse map, cart, checkout, profile, subscriptions")
    ContainerDb(postgres, "PostgreSQL 15", "Database", "All domain aggregates: Vendor, Bouquet, Order, Cart, Customer, SmartVase, etc.")
    ContainerDb(redis, "Redis 7", "Cache + Blacklist", "Query cache (versioned), token blacklist, idempotency keys")
    Container(keycloak, "Keycloak 23", "Identity Provider", "OIDC, JWT issuance, user management, email verification")
    Container(rabbitmq, "RabbitMQ", "Message Broker", "Domain event bus (InMemoryEventBus currently active)")
    Container(mqtt, "Mosquitto MQTT", "IoT Broker", "Vase telemetry, heartbeat, command dispatch")
    Container(signalr, "SignalR Hub (in-process)", "WebSocket Hub (inside REST API process)", "BouquetHub — real-time bouquet updates, order tracking; runs in-process with the REST API")
  }

  System_Ext(stripe, "Stripe")
  System_Ext(fcm, "Firebase FCM")
  System_Ext(twilio, "Twilio")
  System_Ext(nominatim, "Nominatim OSM")
